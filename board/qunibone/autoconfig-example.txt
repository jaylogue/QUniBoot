# QUniBoot Auto-configuration Settings File
#
# Use this file to automatically configure various OS settings when the 
# QUniBoot system boots. Auto-configuration can be performed at first
# boot of a newly installed SD card, or on any subsequent boot.
#
# To enable auto-configuration:
#
# - mount the QUniBoot SD card on your computer and edit the settings
#   below as needed for your system. Uncomment the settings you wish
#   to change and leave others commented out.
#
#     Note that all settings values are optional. Any value that is not
#     set will leave the existing system configuration unchanged.
#
# - When done editing, rename the file to "autoconfig.txt" and eject the
#   SD card.
#
# - Insert the card into the BeagleBone Black and boot the system using
#   the updated SD card.
#
# During the boot process, the system will automatically apply the given
# settings. When the process completes, the system will flash the UniBone/
# QBone LEDs 3 times. If an error occurs, the system will light the LEDs
# and leave them on. The system log file (/var/log/messages) can be
# inspected to determine the cause of any errors.
#
# After auto-configuration completes, the "autoconfig.txt" file will be 
# renamed to "autoconfig-completed.txt". You can rename this file back
# to "autoconfig.txt" to perform auto-configuration again.
#

# Set the system hostname
#
#HOSTNAME=<string>

# Configure the system to use a static IPv4 address, netmask, gateway
# and DNS server.
#
# Note that CIDR notation (e.g. XXX.XXX.XXX.XXX/24) CANNOT be used in
# the ip_addr setting. Use netmask instead.
#
#IP_ADDR=<XXX.XXX.XXX.XXX>
#NETMASK=<XXX.XXX.XXX.XXX>
#GATEWAY=<XXX.XXX.XXX.XXX>
#NAMESERVER=<XXX.XXX.XXX.XXX>

# Configure the system to use DHCP to acquire network configuration.
#
# This is the default, but enabling this line will reset an existing
# static IP address configuration back to using DHCP.
#
#IP_ADDR=DHCP

# Set the password for the root user.
#
#ROOT_PASSWORD=<string>

# Add the given SSH public key to the root user's authorized_keys file
#
# The value should be the base64 encoded public key in standard authorized_keys
# format (e.g. "ssh-rsa AAA....").
#
# This setting has no effect if the key already exists in root's
# authorized_keys file.
#
#ROOT_AUTHORIZED_KEY=<base64-pub-key>

# Clear (delete) the root user's authorized_keys file
#
# This can be used with the ROOT_AUTHORIZED_KEY setting to entirely
# replace the root user's authorized_keys file.
#
#ROOT_AUTHORIZED_KEY=CLEAR

# Install SSH host key files
#
# Key files must be located in the same directory as the autoconfig.txt
# file and should be named ssh_host_*_key and ssh_host_*_key.pub (e.g.
# ssh_host_rsa_key, ssh_host_rsa_key.pub). Existing key files on the
# system, if any, are preserved as .orig files.
#
# This setting has no effect if the contents of the new files are the same
# as the files already installed on the system.
#
#INSTALL_HOST_KEYS=TRUE
